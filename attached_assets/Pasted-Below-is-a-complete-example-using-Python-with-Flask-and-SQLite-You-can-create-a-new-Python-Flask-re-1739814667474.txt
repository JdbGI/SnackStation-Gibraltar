Below is a complete example using Python with Flask and SQLite. You can create a new Python Flask repl on Replit and add the following files. This example provides:
	•	A login page (used by both admin and partners)
	•	An admin dashboard with links to add a new partner and add machine data
	•	A partner dashboard that shows month‐by‐month machine data

	Note: This example is basic and does not cover all production-level security or error checking. You may need to adapt it further for your needs.

1. Create a File Called app.py

from flask import Flask, render_template, request, redirect, url_for, session, flash
import sqlite3
from werkzeug.security import generate_password_hash, check_password_hash

app = Flask(__name__)
app.secret_key = 'your_secret_key_here'  # Replace with a secure key

def get_db_connection():
    conn = sqlite3.connect('database.db')
    conn.row_factory = sqlite3.Row
    return conn

def init_db():
    conn = get_db_connection()
    cursor = conn.cursor()
    # Create the users table
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS users (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            username TEXT NOT NULL UNIQUE,
            password TEXT NOT NULL,
            role TEXT NOT NULL
        )
    ''')
    # Create the machine_data table
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS machine_data (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            partner_id INTEGER,
            month TEXT,
            sales INTEGER,
            commission REAL,
            FOREIGN KEY (partner_id) REFERENCES users(id)
        )
    ''')
    conn.commit()
    conn.close()

# Initialise the database
init_db()

@app.route('/')
def index():
    if 'user_id' in session:
        if session.get('role') == 'admin':
            return redirect(url_for('admin_dashboard'))
        elif session.get('role') == 'partner':
            return redirect(url_for('partner_dashboard'))
    return redirect(url_for('login'))

@app.route('/login', methods=['GET', 'POST'])
def login():
    if request.method == 'POST':
        username = request.form['username']
        password = request.form['password']
        
        conn = get_db_connection()
        user = conn.execute('SELECT * FROM users WHERE username = ?', (username,)).fetchone()
        conn.close()
        
        if user and check_password_hash(user['password'], password):
            session['user_id'] = user['id']
            session['username'] = user['username']
            session['role'] = user['role']
            flash('Logged in successfully!', 'success')
            if user['role'] == 'admin':
                return redirect(url_for('admin_dashboard'))
            else:
                return redirect(url_for('partner_dashboard'))
        else:
            flash('Invalid username or password', 'error')
    return render_template('login.html')

@app.route('/logout')
def logout():
    session.clear()
    flash('You have been logged out', 'info')
    return redirect(url_for('login'))

@app.route('/admin_dashboard')
def admin_dashboard():
    if 'user_id' not in session or session.get('role') != 'admin':
        flash('You need to be logged in as an admin to access this page', 'error')
        return redirect(url_for('login'))
    return render_template('admin_dashboard.html')

@app.route('/partner_dashboard')
def partner_dashboard():
    if 'user_id' not in session or session.get('role') != 'partner':
        flash('You need to be logged in as a partner to access this page', 'error')
        return redirect(url_for('login'))
    
    partner_id = session.get('user_id')
    conn = get_db_connection()
    data = conn.execute('SELECT * FROM machine_data WHERE partner_id = ?', (partner_id,)).fetchall()
    conn.close()
    
    return render_template('partner_dashboard.html', data=data)

@app.route('/add_partner', methods=['GET', 'POST'])
def add_partner():
    if 'user_id' not in session or session.get('role') != 'admin':
        flash('You need to be logged in as an admin to access this page', 'error')
        return redirect(url_for('login'))
    
    if request.method == 'POST':
        username = request.form['username']
        password = request.form['password']
        
        hashed_password = generate_password_hash(password)
        conn = get_db_connection()
        try:
            conn.execute('INSERT INTO users (username, password, role) VALUES (?, ?, ?)',
                         (username, hashed_password, 'partner'))
            conn.commit()
            flash('Partner added successfully!', 'success')
        except sqlite3.IntegrityError:
            flash('Username already exists', 'error')
        conn.close()
        return redirect(url_for('admin_dashboard'))
    
    return render_template('add_partner.html')

@app.route('/add_machine_data', methods=['GET', 'POST'])
def add_machine_data():
    if 'user_id' not in session or session.get('role') != 'admin':
        flash('You need to be logged in as an admin to access this page', 'error')
        return redirect(url_for('login'))
    
    if request.method == 'POST':
        partner_username = request.form['partner_username']
        month = request.form['month']
        sales = request.form['sales']
        commission = request.form['commission']
        
        conn = get_db_connection()
        partner = conn.execute('SELECT * FROM users WHERE username = ? AND role = ?', (partner_username, 'partner')).fetchone()
        if partner:
            conn.execute('INSERT INTO machine_data (partner_id, month, sales, commission) VALUES (?, ?, ?, ?)',
                         (partner['id'], month, sales, commission))
            conn.commit()
            flash('Machine data added successfully!', 'success')
        else:
            flash('Partner not found', 'error')
        conn.close()
        return redirect(url_for('admin_dashboard'))
    
    return render_template('add_machine_data.html')

if __name__ == '__main__':
    # Replit typically runs on port 3000
    app.run(host='0.0.0.0', port=3000, debug=True)

4. Running the Application on Replit
	1.	Ensure Flask is installed:
In your requirements.txt, add:

Flask==2.2.5

(Adjust the version as needed.)

	2.	Run the App:
Replit usually uses port 3000. The code is set to run on host='0.0.0.0' and port=3000, so simply click the run button on Replit.
	3.	Using the Dashboard:
	•	Navigate to the login page.
	•	For testing, you might need to manually add an admin user to your database (for example, by temporarily adding a route or by using the SQLite shell).
	•	Once logged in as an admin, you can add new partner accounts and machine data.
	•	Partners can then log in to view their monthly sales and commission data.

This should provide a solid starting point for your management dashboard. You can then instruct Replit’s AI agent to create and manage these files if needed by providing the code blocks and file names.
